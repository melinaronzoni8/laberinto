import * as THREE from 'three';
import { escalatorImages } from './images.js';

    // Laterales cruzadas

    // Segundo nivel suspendido en el aire

    { id: 6,  x: -9,  y: 7,  z: -22, rotY: Math.PI / 4,     length: 15, width: 2.0, direction: 1,  speed: 1.3 },

    { id: 14, x: 15,  y: 18, z: -44, rotY: Math.PI / 3,     length: 16, width: 2.2, direction: -1, speed: 1.2 }

export class EscalatorSystem {
    constructor(scene) {
        this.scene = scene;
        this.escalators = [];
        this.textureLoader = new THREE.TextureLoader();

        // Materiales brutalistas compartidos
        this.concreteMat = new THREE.MeshStandardMaterial({
            color: 0x555555,
            roughness: 0.9,
            metalness: 0.1
        });

        this.darkMetalMat = new THREE.MeshStandardMaterial({
            color: 0x222222,
            roughness: 0.4,
            metalness: 0.8
        });

        this.stepMat = new THREE.MeshStandardMaterial({
            color: 0x3a3a3a,
            roughness: 0.35,
            metalness: 0.85
        });

        this.glassMat = new THREE.MeshPhysicalMaterial({
            color: 0xaaccbb,
            transparent: true,
            opacity: 0.28,
            roughness: 0.1,
            metalness: 0.1,
            side: THREE.DoubleSide,
            depthWrite: false
        });

        this.buildAll();
    }

    buildAll() {
        escalatorConfigs.forEach(cfg => {
            const group = new THREE.Group();
            group.position.set(cfg.x, cfg.y, cfg.z);
            group.rotation.y = cfg.rotY;

            const angle = Math.PI / 6; // Inclinación estándar de 30°
            const height = Math.sin(angle) * cfg.length;
            const run = Math.cos(angle) * cfg.length;

            // 1. Base / Chasis de hormigón brutalista
            const beamGeo = new THREE.BoxGeometry(cfg.width + 0.4, 0.8, cfg.length);
            const beam = new THREE.Mesh(beamGeo, this.concreteMat);
            beam.position.set(0, height / 2 - 0.3, -run / 2);
            beam.rotation.x = angle;
            beam.castShadow = true;
            beam.receiveShadow = true;
            group.add(beam);

            // Plataformas de embarque (inferior y superior)
            const landingGeo = new THREE.BoxGeometry(cfg.width + 0.4, 0.8, 2.2);
            const bottomLanding = new THREE.Mesh(landingGeo, this.concreteMat);
            bottomLanding.position.set(0, -0.3, 1.0);
            bottomLanding.receiveShadow = true;
            group.add(bottomLanding);

            const topLanding = new THREE.Mesh(landingGeo, this.concreteMat);
            topLanding.position.set(0, height - 0.3, -run - 1.0);
            topLanding.receiveShadow = true;
            group.add(topLanding);

            // 2. Escalones móviles individuales
            const stepCount = Math.floor(cfg.length * 2.2);
            const steps = [];
            const stepGeo = new THREE.BoxGeometry(cfg.width - 0.15, 0.18, 0.45);

            for (let i = 0; i < stepCount; i++) {
                const stepMesh = new THREE.Mesh(stepGeo, this.stepMat);
                stepMesh.castShadow = true;
                stepMesh.receiveShadow = true;
                group.add(stepMesh);
                steps.push({
                    mesh: stepMesh,
                    progress: i / stepCount // Posición normalizada entre 0 y 1
                });
            }

            // 3. Barandas de vidrio + Pasamanos negros + Espacio para PNG
            const railHeight = 1.05;
            const animatedTextures = [];
            const imgConfig = escalatorImages.find(item => item.escalatorId === cfg.id);

            [-1, 1].forEach(sideSign => {
                const sideName = sideSign === -1 ? "left" : "right";
                const xOffset = sideSign * (cfg.width / 2 + 0.05);

                // Panel de vidrio inclinado
                const glassGeo = new THREE.PlaneGeometry(cfg.length, railHeight);
                const glassMesh = new THREE.Mesh(glassGeo, this.glassMat);
                glassMesh.position.set(xOffset, height / 2 + railHeight / 2, -run / 2);
                glassMesh.rotation.y = Math.PI / 2;
                glassMesh.rotation.x = -angle; // Alinear con la pendiente
                // Ajuste de rotación combinada usando contenedor
                const railContainer = new THREE.Group();
                railContainer.position.set(xOffset, height / 2 + railHeight / 2, -run / 2);
                railContainer.rotation.y = Math.PI / 2;
                railContainer.rotation.z = angle;

                const cleanGlass = new THREE.Mesh(glassGeo, this.glassMat);
                railContainer.add(cleanGlass);

                // Pasamanos superior oscuro
                const handrailGeo = new THREE.BoxGeometry(cfg.length + 0.4, 0.1, 0.14);
                const handrail = new THREE.Mesh(handrailGeo, this.darkMetalMat);
                handrail.position.set(0, railHeight / 2 + 0.04, 0);
                railContainer.add(handrail);

                // 4. Aplicar imagen PNG si está configurada para esta escalera y lado
                if (imgConfig && (imgConfig.side === "both" || imgConfig.side === sideName)) {
                    const texture = this.textureLoader.load(
                        imgConfig.image,
                        (tex) => {
                            tex.wrapS = THREE.RepeatWrapping;
                            tex.wrapT = THREE.ClampToEdgeWrapping;
                            tex.repeat.set(imgConfig.repeatX || 1, 1);
                            tex.colorSpace = THREE.SRGBColorSpace;
                        },
                        undefined,
                        () => {
                            // Si el archivo PNG aún no existe en assets/, no genera error crítico
                            console.warn(`Falta cargar el archivo: ${imgConfig.image}`);
                        }
                    );

                    const pngMat = new THREE.MeshBasicMaterial({
                        map: texture,
                        transparent: true,
                        side: THREE.DoubleSide,
                        depthWrite: false
                    });

                    // Plano ligeramente separado del vidrio para evitar parpadeo (z-fighting)
                    const decalGeo = new THREE.PlaneGeometry(cfg.length * 0.92, railHeight * 0.85);
                    const decalMesh = new THREE.Mesh(decalGeo, pngMat);
                    decalMesh.position.set(0, 0, sideSign * 0.015);
                    railContainer.add(decalMesh);

                    if (imgConfig.moveWithBelt) {
                        animatedTextures.push(texture);
                    }
                }

                group.add(railContainer);
            });

            this.scene.add(group);

            this.escalators.push({
                config: cfg,
                steps,
                height,
                run,
                animatedTextures
            });
        });
    }

    update(delta) {
        this.escalators.forEach(esc => {
            const { config, steps, height, run, animatedTextures } = esc;
            const stepDelta = (config.speed * delta * config.direction) / config.length;

            // Mover cada escalón en bucle
            steps.forEach(step => {
                step.progress += stepDelta;
                if (step.progress > 1) step.progress -= 1;
                if (step.progress < 0) step.progress += 1;

                step.mesh.position.set(
                    0,
                    step.progress * height,
                    -step.progress * run
                );
            });

            // Desplazar la textura PNG por la baranda de vidrio al ritmo de la escalera
            animatedTextures.forEach(tex => {
                tex.offset.x -= stepDelta * (tex.repeat.x || 1);
            });
        });
    }
}
